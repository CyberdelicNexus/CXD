**Test 1: Create board, add elements, edit, reload, verify persistence**

```typescript
test.describe('Solo Happy Path', () => {
  test('Create board → add elements → edit → reload → verify persistence', async ({ page }) => {
    test.setTimeout(120_000); // generous timeout for multi-step flow

    // Step 1: Go to dashboard
    await page.goto('/dashboard');
    // Wait for dashboard to load (existing boards visible OR empty state)
    await page.waitForTimeout(3000);

    // Step 2: Create a new board
    // EXECUTOR: Find and click the create board button
    // Wait for navigation to /cxd
    await page.waitForURL('**/cxd', { timeout: 15000 });
    await page.waitForTimeout(4000); // wait for canvas to initialize

    // Step 3: Add 3+ note elements to the canvas
    // EXECUTOR: Find the add element button(s) in toolbar
    // Click add button 3 times with waits between
    for (let i = 0; i < 3; i++) {
      // Click add element button
      await page.waitForTimeout(1000);
    }

    // Step 4: Edit each note card with unique text
    const ts = Date.now();
    const texts = [`solo-card1-${ts}`, `solo-card2-${ts}`, `solo-card3-${ts}`];
    const noteCards = page.locator('.card--note-resizable');
    const count = await noteCards.count();
    console.log(`  Found ${count} note cards on canvas`);

    for (let i = 0; i < Math.min(count, 3); i++) {
      await editNoteTitle(page, i, texts[i]);
      console.log(`  Edited card ${i}: "${texts[i]}"`);
    }

    // Step 5: Wait for save
    await page.waitForTimeout(5000); // debounce + Supabase write

    // Step 6: Hard reload
    await page.reload({ waitUntil: 'domcontentloaded' });
    await page.waitForTimeout(5000); // Yjs load from IndexedDB + Supabase

    // Step 7: Verify all texts persisted
    const body = await page.textContent('body');
    for (const text of texts) {
      if (count > texts.indexOf(text)) {
        expect(body).toContain(text);
        console.log(`  ✓ "${text}" persisted after reload`);
      }
    }

    console.log('✓ Solo happy path: create → add → edit → persist verified');
  });
});
```
