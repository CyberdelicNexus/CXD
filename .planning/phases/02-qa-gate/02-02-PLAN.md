Create `e2e/collab-persistence.spec.ts` with the following structure. This test uses TWO Chrome browser contexts (not two different browser engines), avoiding the Edge dependency.

**Key design decisions:**
- Use `browser.newContext()` twice with different storage states (user-a.json for owner, user-b.json for collaborator)
- Both contexts use Chromium — no Edge requirement
- Target a shared board that both users have access to. Use the board name "Collab Test 2" (same as collab-realtime.spec.ts) since it's already shared between both users.
- If "Collab Test 2" is not found, use the first available board (the test assumes the two users share at least one board)

**Reuse helper patterns from existing tests:**
- `openNamedCanvas()` pattern from `collab-realtime.spec.ts` for finding a board by name
- `typeOnNoteCard()` pattern from `collab-realtime.spec.ts` for editing note cards
- `waitForLiveText()` pattern from `collab-realtime.spec.ts` for polling real-time sync
- `pageContains()` and `hardReload()` patterns from `persistence.spec.ts`

**Test implementation:**

```typescript
import { test, expect, Page, BrowserContext } from '@playwright/test';
import path from 'path';

const BASE_URL = 'http://localhost:3000';
const AUTH_A = path.join(__dirname, '.auth/user-a.json');
const AUTH_B = path.join(__dirname, '.auth/user-b.json');
const BOARD_NAME = 'Collab Test 2';
const REALTIME_TIMEOUT = 20_000;

// ─── Helpers ───

// Copy openNamedCanvas from collab-realtime.spec.ts (navigates to dashboard,
// finds board by name, clicks to open, waits for canvas to load)

// Copy typeOnNoteCard from collab-realtime.spec.ts (double-clicks note card,
// fills title input, clicks outside to blur)

// Copy waitForLiveText from collab-realtime.spec.ts (polls page body for text)

async function hardReload(page: Page): Promise<void> {
  await page.reload({ waitUntil: 'domcontentloaded' });
  await page.waitForTimeout(5000); // Yjs load from IndexedDB + Supabase
}

async function pageContains(page: Page, text: string): Promise<boolean> {
  const body = await page.textContent('body');
  return body?.includes(text) ?? false;
}

// ─── Tests ───

test.describe('Collaboration with Persistence', () => {

  test('Collaborator edits → both see live → both reload → changes persist', async ({ browser }) => {
    test.setTimeout(180_000); // 3 min timeout for full collab flow

    // Create two Chrome contexts with different auth states
    const ctxOwner: BrowserContext = await browser.newContext({
      storageState: AUTH_A,
      viewport: { width: 1280, height: 800 },
    });
    const ctxCollab: BrowserContext = await browser.newContext({
      storageState: AUTH_B,
      viewport: { width: 1280, height: 800 },
    });

    const pageOwner: Page = await ctxOwner.newPage();
    const pageCollab: Page = await ctxCollab.newPage();

    try {
      // ── Step 1: Both open the shared board ──
      console.log('\n── Step 1: Opening shared canvas ──');
      await openNamedCanvas(pageOwner, BOARD_NAME);
      await openNamedCanvas(pageCollab, BOARD_NAME);

      // Let Yjs channels establish
      await Promise.all([
        pageOwner.waitForTimeout(3000),
        pageCollab.waitForTimeout(3000),
      ]);

      // ── Step 2: Collaborator edits a note card ──
      console.log('\n── Step 2: Collaborator edits ──');
      const collabText = await typeOnNoteCard(pageCollab, 'collab-persist');
      console.log(`  Collaborator typed: "${collabText}"`);

      // ── Step 3: Verify owner sees the edit in real-time ──
      console.log('\n── Step 3: Verify real-time sync to owner ──');
      const ownerSeesIt = await waitForLiveText(pageOwner, collabText, 'Owner');
      expect(ownerSeesIt, `Owner should see collaborator's edit "${collabText}" in real-time`).toBe(true);

      // ── Step 4: Wait for saves to complete ──
      console.log('\n── Step 4: Waiting for persistence ──');
      await Promise.all([
        pageOwner.waitForTimeout(6000),  // debounce + Supabase write
        pageCollab.waitForTimeout(6000),
      ]);

      // ── Step 5: Both users reload ──
      console.log('\n── Step 5: Both users reload ──');
      await Promise.all([
        hardReload(pageOwner),
        hardReload(pageCollab),
      ]);

      // ── Step 6: Verify collaborator's edit persisted for BOTH users ──
      console.log('\n── Step 6: Verify persistence after reload ──');
      const ownerBody = await pageOwner.textContent('body');
      const collabBody = await pageCollab.textContent('body');

      expect(ownerBody).toContain(collabText);
      console.log(`  ✓ Owner sees "${collabText}" after reload`);

      expect(collabBody).toContain(collabText);
      console.log(`  ✓ Collaborator sees "${collabText}" after reload`);

      console.log('\n✓ Collaboration persistence verified: edit → sync → reload → persisted');

    } finally {
      await ctxOwner.close();
      await ctxCollab.close();
    }
  });

  test('Bidirectional edits persist after reload', async ({ browser }) => {
    test.setTimeout(180_000);

    const ctxOwner = await browser.newContext({
      storageState: AUTH_A,
      viewport: { width: 1280, height: 800 },
    });
    const ctxCollab = await browser.newContext({
      storageState: AUTH_B,
      viewport: { width: 1280, height: 800 },
    });

    const pageOwner = await ctxOwner.newPage();
    const pageCollab = await ctxCollab.newPage();

    try {
      // Both open shared board
      await openNamedCanvas(pageOwner, BOARD_NAME);
      await openNamedCanvas(pageCollab, BOARD_NAME);
      await Promise.all([
        pageOwner.waitForTimeout(3000),
        pageCollab.waitForTimeout(3000),
      ]);

      // Owner edits one card
      const ownerText = await typeOnNoteCard(pageOwner, 'owner-bidi');
      console.log(`  Owner typed: "${ownerText}"`);

      // Wait for sync to propagate
      await Promise.all([
        pageOwner.waitForTimeout(2000),
        pageCollab.waitForTimeout(2000),
      ]);

      // Collaborator edits a DIFFERENT card (use last card if multiple exist)
      // First check how many note cards exist
      const noteCards = pageCollab.locator('.card--note-resizable');
      const cardCount = await noteCards.count();

      let collabText: string;
      if (cardCount > 1) {
        // Edit the last card (different from owner's first card)
        collabText = `collab-bidi-${Date.now()}`;
        const lastCard = noteCards.last();
        await lastCard.dblclick({ delay: 100 });
        const input = lastCard.locator('input').first();
        await input.waitFor({ state: 'visible', timeout: 8000 });
        await input.click({ clickCount: 3 });
        await input.fill(collabText);
        await pageCollab.mouse.click(10, 10);
        await pageCollab.waitForTimeout(500);
      } else {
        // Only one card — collaborator edits the TipTap body instead of title
        collabText = `collab-bidi-${Date.now()}`;
        const noteCard = noteCards.first();
        await noteCard.dblclick({ delay: 100 });
        const editable = pageCollab.locator('[contenteditable]').first();
        await editable.waitFor({ state: 'visible', timeout: 8000 });
        await editable.click({ clickCount: 3 });
        await editable.fill(collabText);
        await pageCollab.mouse.click(10, 10);
        await pageCollab.waitForTimeout(500);
      }
      console.log(`  Collaborator typed: "${collabText}"`);

      // Wait for saves
      await Promise.all([
        pageOwner.waitForTimeout(6000),
        pageCollab.waitForTimeout(6000),
      ]);

      // Both reload
      await Promise.all([
        hardReload(pageOwner),
        hardReload(pageCollab),
      ]);

      // Verify BOTH edits persisted in BOTH tabs
      const ownerBody = await pageOwner.textContent('body');
      const collabBody = await pageCollab.textContent('body');

      expect(ownerBody).toContain(ownerText);
      expect(collabBody).toContain(ownerText);
      console.log(`  ✓ Owner's edit "${ownerText}" visible to both after reload`);

      expect(ownerBody).toContain(collabText);
      expect(collabBody).toContain(collabText);
      console.log(`  ✓ Collaborator's edit "${collabText}" visible to both after reload`);

      console.log('\n✓ Bidirectional edits both persisted after reload');

    } finally {
      await ctxOwner.close();
      await ctxCollab.close();
    }
  });
});
```

**Critical implementation notes for the executor:**
- Copy the helper functions (`openNamedCanvas`, `typeOnNoteCard`, `waitForLiveText`) from `e2e/collab-realtime.spec.ts` — do NOT import them (each spec file should be self-contained)
- Use `browser.newContext()` for both users (both are Chromium) — do NOT use `chromium.launch()` for a separate browser
- Both contexts navigate to the same board by name ("Collab Test 2")
- The test must work with `workers: 1` and `fullyParallel: false` (existing config)
- Use generous timeouts: Yjs channel establishment takes ~3-8s, save debounce is 2s, Supabase write adds 1-2s
- The `hardReload` wait should be 5s (not 4s like persistence.spec.ts) because collaborator contexts may take longer to reload from Supabase without IndexedDB cache
- If the "Collab Test 2" board doesn't exist, the test should still work by falling back to the first available board (the `openNamedCanvas` helper already does this)
